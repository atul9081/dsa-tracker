### 2.4 Update Queue Generation to Use `user_settings`

**File:** `app/api/queue/route.ts`

`generateQueue()` currently reads `QUEUE_CONFIG` (hardcoded). New flow:

```ts
async function generateQueue(supabase, userId, date) {
  // 1. Fetch user settings (fallback to QUEUE_CONFIG defaults)
  const { data: settings } = await supabase
    .from('user_settings')
    .select('*')
    .eq('user_id', userId)
    .single();

  const cfg = {
    dailyCount: settings?.daily_count ?? QUEUE_CONFIG.dailyCount,
    hardCount: settings?.hard_count ?? QUEUE_CONFIG.hardCount,
    mediumCount: settings?.medium_count ?? QUEUE_CONFIG.mediumCount,
    easyCount: settings?.easy_count ?? QUEUE_CONFIG.easyCount,
    enabledTopicSlugs: settings?.enabled_topic_slugs ?? null, // null = all
    topicWeightOverrides: settings?.topic_weight_overrides ?? {},
  };

  // 2. Fetch unsolved problems
  let pool = await fetchUnsolvedProblems(supabase);

  // 3. Filter by enabled topics (if Focus Mode is set)
  if (cfg.enabledTopicSlugs) {
    pool = pool.filter(p =>
      p.topic_tags.some(t => cfg.enabledTopicSlugs.includes(t))
    );
  }

  // 4. Topic-normalized 2-stage sampling (fix data skew)
  //    Stage 1: pick a topic (weighted by olt_score x topic_weight_override)
  //    Stage 2: pick a problem from that topic (weighted by difficulty multiplier)
  //    Guard: no topic more than 3 times in a single queue

  // ... (implementation detail below)
}
```

**Topic-normalized sampling algorithm:**

```ts
function topicNormalizedSample(pool, count, cfg) {
  const topicGroups = groupByPrimaryTopic(pool); // primary = highest olt_score tag
  const topicCounts: Record<string, number> = {};
  const selected: ProblemWithStatus[] = [];

  for (let i = 0; i < count && Object.keys(topicGroups).length > 0; i++) {
    // Filter out topics already at max (3)
    const eligible = Object.entries(topicGroups).filter(
      ([slug]) => (topicCounts[slug] ?? 0) < 3
    );
    if (eligible.length === 0) break;

    // Weighted topic pick
    const topicWeights = eligible.map(([slug, problems]) => {
      const baseScore = TOPIC_SCORE_MAP[slug] ?? 1;
      const override = cfg.topicWeightOverrides[slug] ?? 1;
      return Math.pow(baseScore * override, 2);
    });
    const pickedTopic = weightedPick(eligible, topicWeights);

    // Weighted problem pick from that topic
    const problems = topicGroups[pickedTopic];
    const prob = weightedPick(problems, problems.map(p =>
      QUEUE_CONFIG.difficultyMultiplier[p.difficulty] ?? 1
    ));

    selected.push(prob);
    topicCounts[pickedTopic] = (topicCounts[pickedTopic] ?? 0) + 1;

    // Remove from pool to avoid duplicates
    topicGroups[pickedTopic] = topicGroups[pickedTopic].filter(p => p.id !== prob.id);
    if (topicGroups[pickedTopic].length === 0) delete topicGroups[pickedTopic];
  }

  return selected;
}
```

Note: We still respect hardCount/mediumCount splits — the 2-stage sampling is applied *within* each difficulty bucket.
