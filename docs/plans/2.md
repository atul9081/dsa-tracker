### 1.1 Database Migration (run in Supabase SQL Editor)

Run in this exact order:

```sql
-- Step 1: Add user_id to user_progress
ALTER TABLE user_progress
  ADD COLUMN user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid();

ALTER TABLE user_progress DROP CONSTRAINT user_progress_pkey;
ALTER TABLE user_progress ADD PRIMARY KEY (problem_id, user_id);

-- Step 2: Add user_id to daily_queues
ALTER TABLE daily_queues
  ADD COLUMN user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE DEFAULT auth.uid();

ALTER TABLE daily_queues DROP CONSTRAINT daily_queues_pkey;
ALTER TABLE daily_queues ADD PRIMARY KEY (date, user_id);

-- Step 3: Assign existing rows to you (run after getting your user_id from auth.users)
-- UPDATE user_progress SET user_id = '<your-uuid>' WHERE user_id IS NULL;
-- UPDATE daily_queues   SET user_id = '<your-uuid>' WHERE user_id IS NULL;
-- (I will provide this after you create your account)

-- Step 4: Drop and recreate the view (now per-user via auth.uid())
DROP VIEW problems_with_status;
CREATE VIEW problems_with_status AS
  SELECT p.*,
    COALESCE(up.status, 'TO_DO') AS status,
    up.updated_at AS status_updated_at
  FROM problems p
  LEFT JOIN user_progress up
    ON p.id = up.problem_id AND up.user_id = auth.uid();

-- Step 5: Enable RLS
ALTER TABLE problems     ENABLE ROW LEVEL SECURITY;
ALTER TABLE user_progress ENABLE ROW LEVEL SECURITY;
ALTER TABLE daily_queues  ENABLE ROW LEVEL SECURITY;

-- Step 6: Policies
-- problems: anyone authenticated can read
CREATE POLICY "problems_read" ON problems
  FOR SELECT USING (true);

-- user_progress: users only see/write their own rows
CREATE POLICY "own_progress" ON user_progress
  USING (user_id = auth.uid())
  WITH CHECK (user_id = auth.uid());

-- daily_queues: users only see/write their own queues
CREATE POLICY "own_queues" ON daily_queues
  USING (user_id = auth.uid())
  WITH CHECK (user_id = auth.uid());
```
